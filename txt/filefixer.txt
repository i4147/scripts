import argparse
import concurrent.futures
import sys
from pathlib import Path
from dh import get_files
try:
    import puremagic
except ImportError:
    print("Error: This script requires the 'puremagic' library.")
    print('Please install it by running: pip install puremagic')
    sys.exit(1)
def a(b):
    c = {'image/jpeg': '.jpg', 'image/png': '.png', 'image/gif': '.gif', 'image/webp': '.webp', 'image/tiff': '.tiff', 'image/bmp': '.bmp', 'application/pdf': '.pdf', 'application/zip': '.zip', 'application/x-tar': '.tar', 'application/gzip': '.gz', 'application/x-7z-compressed': '.7z', 'application/x-rar': '.rar', 'video/mp4': '.mp4', 'video/x-msvideo': '.avi', 'video/quicktime': '.mov', 'audio/mpeg': '.mp3', 'audio/wav': '.wav', 'text/plain': '.txt', 'application/octet-stream': ''}
    return c.get(b, '')
def d(e):
    try:
        f = puremagic.magic_string(e.read_bytes())
        if f:
            g = f[0]
            ext = g.extension
            if ext and ext != '':
                return ext if ext.startswith('.') else f'.{ext}'
            h = a(g.mime_type)
            if h:
                return h
    except puremagic.main.PureError:
        pass
    except Exception as e:
        print(f'Error reading {e}: {e}')
    return ''
def i(e):
    j = e.suffix.lower()
    if not j:
        return None
    k = d(e)
    if not k:
        return None
    k = k.lower()
    if j != k:
        l = {'.jpeg': '.jpg', '.htm': '.html', '.tif': '.tiff'}
        m = l.get(j, j)
        n = l.get(k, k)
        if m != n:
            return (e, j, k)
    return None
def o(e, j, k):
    p = e.stem + k
    q = e.with_name(p)
    r = 1
    s = q
    while q.exists() and q != e:
        q = s.with_name(f'{s.stem}_{r}{k}')
        r += 1
    if q != e:
        e.rename(q)
        return q
    return e
def t():
    u = argparse.ArgumentParser(description='Recursively detect and optionally fix file extension mismatches based on file headers (magic numbers).')
    u.add_argument('directory', type=str, help='The directory to recursively scan.')
    u.add_argument('-a', '--autofix', action='store_true', help='Automatically rename files to fix detected mismatches in place.')
    u.add_argument('-w', '--workers', type=int, default=4, help='Number of parallel worker threads (default: 4).')
    v = u.parse_args()
    w = Path(v.directory)
    if not w.is_dir():
        print(f"Error: '{w}' is not a valid directory.")
        sys.exit(1)
    print(f"Scanning '{w}' for extension mismatches...")
    if v.autofix:
        print('Autofix is ENABLED. Files will be renamed.')
    cwd = Path.cwd()
    x = get_files(cwd)
    y = []
    with concurrent.futures.ThreadPoolExecutor(max_workers=v.workers) as z:
        a1 = {z.submit(i, f): f for f in x}
        for b1 in concurrent.futures.as_completed(a1):
            c1 = b1.result()
            if c1:
                y.append(c1)
    if not y:
        print('No extension mismatches found!')
        return
    print(f'\nFound {len(y)} mismatches:')
    for e, j, k in sorted(y):
        print(f"[MISMATCH] '{e}' | Current: '{j}' | Detected: '{k}'")
        if v.autofix:
            try:
                q = o(e, j, k)
                if q != e:
                    print(f"  -> Fixed: Renamed to '{q.name}'")
                else:
                    print('  -> Skipped fix: Filename collision or identical.')
            except Exception as e:
                print(f'  -> Error fixing file: {e}')
if __name__ == '__main__':
    raise SystemExit(t())
