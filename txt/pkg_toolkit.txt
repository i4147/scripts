# --- File: check_system_bin.py ---
import hashlib
import shutil
from pathlib import Path
def a(b, c=8192):
    d = hashlib.sha256()
    try:
        with b.open('rb') as f:
            for e in iter(lambda: f.read(c), b''):
                d.update(e)
        return d.hexdigest()
    except (OSError, PermissionError):
        return None
def f():
    g = Path('/system/bin')
    if not g.exists():
        print('⚠️  /system/bin directory not found!')
        return {}
    h = {}
    print('📂 Scanning /system/bin files...')
    for b in g.iterdir():
        try:
            if b.is_file() or b.is_symlink():
                i = a(b)
                if i:
                    h[i] = b.name
        except (PermissionError, OSError):
            continue
    print(f'✅ Scanned {len(h)} files in /system/bin\n')
    return h
def j(k):
    l = Path.cwd()
    m = l / 'matched_system_files'
    m.mkdir(exist_ok=True)
    n = []
    o = []
    print('🔍 Scanning current directory...')
    for b in l.iterdir():
        try:
            if b.is_file() and (not b.name.startswith('.')):
                if b.resolve() == m.resolve():
                    continue
                i = a(b)
                if i and i in k:
                    p = k[i]
                    n.append((b.name, p))
                    if b.name == p:
                        q = m / b.name
                        r = 1
                        s = q
                        while q.exists():
                            q = s.parent / f'{s.stem}_{r}{s.suffix}'
                            r += 1
                        shutil.move(b, q)
                        o.append((b.name, q.name))
                        print(f'  📦 Moved: {b.name} -> {q.name}')
                    else:
                        print(f'  ⚠️  Hash matches but filename differs: {b.name} (system: {p})')
        except (PermissionError, OSError) as e:
            print(f'  ⚠️  Error with {b.name}: {e}')
            continue
    return (n, o)
def t():
    print('-' * 40)
    print('🔐 File Hash Comparison & Move Tool')
    print('-' * 40)
    k = f()
    if not k:
        print('❌ No readable files found in /system/bin')
        return
    n, o = j(k)
    print('\n' + '=' * 40)
    print('📊 SUMMARY')
    print('-' * 40)
    if n:
        print(f'⚠️  Found {len(n)} files with matching hashes:')
        for u, v in n:
            w = '✅ MOVED' if u == v else '❌ Name mismatch'
            print(f'  • {u} matches /system/bin/{v} - {w}')
        if o:
            print(f"\n📦 Moved {len(o)} files to './matched_system_files/' directory:")
            for x, y in o:
                print(f'  • {x} -> {y}')
    else:
        print('✅ No matching files found.')
    print('-' * 40)
if __name__ == '__main__':
    raise SystemExit(t())
