def a():

            for f in d:
                if f.strip():

def h(i, j, k):
    try:
        src = Path(i)
        if not src.is_absolute():
            src = j / src
        if src.exists() and src.is_file():
            rel = src.relative_to(j)
            dst = k / rel
            dst.parent.mkdir(parents=True, exist_ok=True)

        n = pkg.replace('-', '_')
        o = set(j.glob(f'{n}-*.dist-info')) | set(j.glob(f'{pkg}-*.dist-info'))
        for b in o:
            c = b / 'RECORD'
            if c.exists():
                for f in c.read_text().splitlines():
                    if f.strip():
                        p = f.split(',')[0]
                        if p and (not p.endswith('.pyc')):
                            m.append(p)
