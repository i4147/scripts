def a(b):
    with open(b, 'r', encoding='utf-8') as f:
        c = f.read()
    d = re.search('^version\\s*=\\s*(\\d+)', c, re.MULTILINE)
    e = int(d.group(1)) if d else 3
    f = []
    if e >= 2:
        g = re.split('\\n\\[\\[package\\]\\]\\n', c)
        for h in g[1:]:
            pkg = i(h)
            if pkg:
                f.append(pkg)
    else:
        g = re.split('\\n\\[\\[package\\]\\]\\n', c)
        for h in g[1:]:
            pkg = p(h)
            if pkg:
                f.append(pkg)
    return {'version': e, 'packages': f}
def i(h):
    pkg = {}
    j = re.search('^name\\s*=\\s*"([^"]*)"', h, re.MULTILINE)
    d = re.search('^version\\s*=\\s*"([^"]*)"', h, re.MULTILINE)
    k = re.search('^source\\s*=\\s*"([^"]*)"', h, re.MULTILINE)
    if not j or not d:
        return None
    pkg['name'] = j.group(1)
    pkg['version'] = d.group(1)
    if k:
        pkg['source'] = k.group(1)
    l = []
    m = False
    for n in h.split('\n'):
        if n.strip().startswith('dependencies = ['):
            m = True
            o = re.findall('"([^"]*)"', n)
            l.extend(o)
            if ']' in n:
                m = False
        elif m:
            o = re.findall('"([^"]*)"', n)
            l.extend(o)
            if ']' in n:
                m = False
    if l:
        pkg['dependencies'] = l
    return pkg
def p(h):
    pkg = {}
    j = re.search('^name\\s*=\\s*"([^"]*)"', h, re.MULTILINE)
    d = re.search('^version\\s*=\\s*"([^"]*)"', h, re.MULTILINE)
    if not j or not d:
        return None
    pkg['name'] = j.group(1)
    pkg['version'] = d.group(1)
    l = []
    for n in h.split('\n'):
        q = re.match('^\\s*"([^"]+)\\s+([^"]+)"', n)
        if q:
            l.append(f'{q.group(1)} {q.group(2)}')
    if l:
        pkg['dependencies'] = l
    return pkg
def r(f, s=None, t='0.1.0', u=False):

            for pkg in f[1:]:
                v.append(f'''{pkg['name']} = "{pkg['version']}"''')
