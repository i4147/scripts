def j(k, l):
    if k.shape != l.shape:
        return False
    m = np.abs(k.astype(np.int16) - l.astype(np.int16))
    n = m.mean()
    o = (m > 10).any(axis=-1).mean()
    return n < g and o < h
def p(q):
    r = []
    try:
        with x3.open(q) as img:
            if not hasattr(img, 'n_frames'):
                s = x3.new('RGB', img.size, (255, 255, 255))
                s.paste(img.convert('RGBA'), mask=img.convert('RGBA').split()[3] if img.mode in ('RGBA', 'P') else None)
                r.append(np.asarray(s))
                return r
            s = x3.new('RGB', img.size, (255, 255, 255))
            t = None
            for u in range(img.n_frames):
                img.seek(u)
                v = img.info.get('disposal', 0)
                if v == 3 and t is not None:
                    s = t.copy()
                elif v == 2:
                    s = x3.new('RGB', img.size, (255, 255, 255))
                t = s.copy()
                w = img.convert('RGBA')
                s.paste(w, mask=w.split()[3])
                arr = np.asarray(s.convert('RGB'))
                if r and j(r[-1], arr):
                    log.debug('  Skipping near-duplicate frame %d in %s', u, q.name)
                    continue
                r.append(arr)
    except (UnidentifiedImageError, OSError) as exc:
        log.error('Cannot open %s: %s', q, exc)
    return r
def x(q):
