                with lzma.open(f2, 'rb') as g2, open(f2.with_suffix(''), 'wb') as h2:

            elif f2.suffix == '.7z':
                with py7zr.SevenZipFile(f2, mode='r') as r1:
                    r1.extractall(path=f2.parent)
            elif f2.suffix == '.lz4':
                with lz4.frame.open(f2, mode='rb') as g2, open(f2.with_suffix(''), 'wb') as h2:

            elif f2.suffix == '.br':
                with open(f2, 'rb') as g2:
                    i2 = g2.read()
                    j2 = brotli.decompress(i2)
                    with open(f2.with_suffix(''), 'wb') as h2:
                        h2.write(j2)
            elif f2.suffix == '.zst':
                with open(f2, 'rb') as g2:
                    k2 = k5.ZstdDecompressor()
                    with open(f2.with_suffix(''), 'wb') as h2:
                        k2.copy_stream(g2, h2)
            elif f2.suffix == '.zip' or f2.suffix == '.whl':

        elif f2.suffix in d2:
