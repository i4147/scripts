import re
import sys
from pathlib import Path
def a():
    if len(sys.argv) != 2:
        print('Usage: python check_dups.py <const_file.py>')
        sys.exit(1)
    b = Path(sys.argv[1])
    if not b.exists():
        print(f'File not found: {b}')
        sys.exit(1)
    c = b.parent / 'dup_const.py'
    d = re.compile('^\\s*([a-zA-Z_][a-zA-Z0-9_]*)\\s*=')
    e = b.read_text().splitlines(keepends=True)
    f = {}
    g = []
    for i, h in enumerate(e):
        i = d.match(h)
        if i:
            j = i.group(1)
            if j not in f:
                f[j] = i
            else:
                g.append((i, h))
    if not g:
        print('No duplicates found.')
        return
    k = [h for idx, h in enumerate(e) if idx not in {d[0] for d in g}]
    b.write_text(''.join(k))
    with c.open('a') as f:
        f.write(f'\n# Duplicate declarations from {b.name}\n')
        for _, l in g:
            f.write(l)
    print('Kept the first declaration of each constant.')
    print(f'Moved {len(g)} duplicate declarations to {c}')
    print(f'Updated {b} in place.')
