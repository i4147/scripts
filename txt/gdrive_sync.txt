import io
import os
import pickle
from google.auth.transport.requests import Request
from google_auth_oauthlib.flow import InstalledAppFlow
from googleapiclient.discovery import Resource, build
from googleapiclient.http import MediaIoBaseDownload
a = ['https://www.googleapis.com/auth/drive.readonly']
def b():
    c = None
    if os.path.exists('token.pickle'):
        with open('token.pickle', 'rb') as d:
            c = pickle.load(d)
    if not c or not c.valid:
        if c and c.expired and c.refresh_token:
            c.refresh(Request())
        else:
            e = InstalledAppFlow.from_client_secrets_file('credentials.json', a)
            c = e.run_local_server(port=0)
        with open('token.pickle', 'wb') as d:
            pickle.dump(c, d)
    return build('drive', 'v3', credentials=c)
def f(g, h):
    i = f"name='{h}' and mimeType='application/vnd.google-apps.folder' and trashed=false"
    j = g.files().list(q=i, fields='files(id, name)').execute()
    k = j.get('files', [])
    if not k:
        raise Exception(f"Folder '{h}' not found in Google Drive")
    return k[0]['id']
def l(g, m, n):
    i = f"'{m}' in parents and trashed=false"
    j = g.files().list(q=i, fields='files(id, name, mimeType)').execute()
    k = j.get('files', [])
    for o in k:
        p = os.path.join(n, o['name'])
        if o['mimeType'] == 'application/vnd.google-apps.folder':
            os.makedirs(p, exist_ok=True)
            print(f'Creating folder: {p}')
            l(g, o['id'], p)
        else:
            print(f'Downloading: {p}')
            q = g.files().get_media(fileId=o['id'])
            fh = io.FileIO(p, 'wb')
            r = MediaIoBaseDownload(fh, q)
            s = False
            while not s:
                t, s = r.next_chunk()
                print(f'Download progress: {int(t.progress() * 40)}%')
            fh.close()
def u():
    h = 'notebooks'
    try:
        g = b()
        m = f(g, h)
        print(f"Found folder '{h}' with ID: {m}")
        v = os.path.join(os.getcwd(), h)
        os.makedirs(v, exist_ok=True)
        l(g, m, v)
        print(f"\nSuccessfully downloaded '{h}' to {v}")
    except Exception as e:
        print(f'Error: {e}')
if __name__ == '__main__':
    raise SystemExit(u())
