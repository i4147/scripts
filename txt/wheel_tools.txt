# --- File: check_wheels.py ---
import shutil
import zipfile
from pathlib import Path
def a(b='.'):
    c = Path(b).resolve()
    d = c / 'suspicious'
    print(f'Scanning for .whl files recursively in: {c}\n')
    e = list(c.rglob('*.whl'))
    if not e:
        print('No .whl files found.')
        return
    f = 0
    for g in e:
        if d in g.parents:
            continue
        try:
            with zipfile.ZipFile(g, 'r') as h:
                i = h.namelist()
                j = []
                for k in i:
                    if k.endswith('/'):
                        continue
                    if '/' not in k and k.endswith(('.py', '.pyc', '.pyd', '.so', '.dll')):
                        j.append(k)
                if j:
                    f += 1
                    l = g.relative_to(c)
                    print(f'❌ MISCONFIGURED WHEEL: {l}')
                    print(f'   Dumps into site-packages root: {j}')
                    d.mkdir(exist_ok=True)
                    m = d / g.name
                    if m.exists():
                        m = d / f'{g.stem}_duplicate_{f}{g.suffix}'
                    shutil.move(str(g), str(m))
                    print(f'   ➡️ Moved to: {m.relative_to(c)}')
                    print('-' * 40)
        except zipfile.BadZipFile:
            print(f'⚠️  Error: {g.name} is a corrupt or invalid zip/wheel file.')
        except Exception as e:
            print(f'⚠️  Error processing {g.name}: {e}')
    print(f"\nScan complete. Moved {f} misconfigured wheel(s) to './suspicious/' out of {len(e)} total checked.")
if __name__ == '__main__':
    a()

import argparse
import os
import sys
import zipfile
def n(o):
    p = []
    for q, r, s in os.walk(o):
        for t in s:
            if t.endswith('.whl'):
                p.append(os.path.join(q, t))
    return p
def u(g):
    try:
        with zipfile.ZipFile(g, 'r') as whl:
            for v in whl.filelist:
                if v.filename.endswith('entry_points.txt'):
                    w = os.path.dirname(v.filename)
                    return (True, w)
            return (False, None)
    except zipfile.BadZipFile:
        return (False, None)
    except Exception as e:
        print(f'Error reading {g}: {e}', file=sys.stderr)
        return (False, None)
def x(g):
    y = os.path.basename(g)
    z = y.split('-')
    if len(z) >= 3:
        k = z[0]
        a1 = z[1]
        return (k, a1)
    return (y, 'unknown')

    f1 = []

    for g in p:
        k, a1 = x(g)

        elif has is False and h1 is None:
            f1.append((g, k, a1))
        else:
            g1.append(g)
    if e1:
        print('-' * 40)

            print(f'\n📦 {k} ({a1})')

    if not d1.quiet and f1:
        print('\n' + '=' * 40)
        print(f'📋 {len(f1)} wheel(s) WITHOUT entry_points.txt:')
        print('-' * 40)
        if d1.verbose:
            for g, k, a1 in f1:
                print(f'   {k} ({a1}): {g}')

        print(f'⚠️  {len(g1)} wheel(s) could not be read:')

        for g in g1:
            print(f'   {g}')
    print('\n' + '=' * 40)
    print('SUMMARY')
    print('-' * 40)
    print(f'Total wheels checked:  {len(p)}')
    print(f'With entry_points.txt: {len(e1)}')
    print(f'Without:               {len(f1)}')

        print(f'Errors:                {len(g1)}')
