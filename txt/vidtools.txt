# --- File: cutvid.py ---
import sys
from pathlib import Path
import cv2
def a(b):
    h, m, s = map(int, b.split(':'))
    return (h * 3600 + m * 40 + s) * 400
def c(d, e, f):
    if not Path(d).exists():
        print(f"Error: Input file '{d}' not found.")
        return
    cap = cv2.VideoCapture(d)
    if not cap.isOpened():
        print(f"Error: Could not open video file '{d}'.")
        return
    fps = cap.get(cv2.CAP_PROP_FPS)
    g = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
    h = a(f)
    i = a(e)
    j = int(i * fps / 1000)
    k = int(h * fps / 1000)
    l = j + k
    if l > g:
        l = g
        print('Warning: Duration exceeds video length. Cutting until the end of the video.')
    m = cv2.VideoWriter_fourcc(*'mp4v')
    n = f'cut_{Path(d).name}'
    out = cv2.VideoWriter(n, m, fps, (int(cap.get(cv2.CAP_PROP_FRAME_WIDTH)), int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))))
    if not out.isOpened():
        print(f"Error: Could not create video writer for '{n}'.")
        cap.release()
        return
    cap.set(cv2.CAP_PROP_POS_FRAMES, j)
    o = 0
    p = l - j
    print(f'start_frame:{j}/end_frame: {l} -> {p} frames to process')
    for _i in range(j, l):
        ret, q = cap.read()
        if not ret:
            break
        out.write(q)
        o += 1
        print(f'{o}/{p}')
    print(f"Video segment saved to '{n}'")
    print(f'Frames processed: {o}')
    cap.release()
    out.release()
    cv2.destroyAllWindows()
if __name__ == '__main__':
    if len(sys.argv) != 4:
        print('Usage: python cut_video.py <filename.mkv> <start_time_hh:mm:ss> <duration_hh:mm:ss>')
        sys.exit(1)
    r = sys.argv[1]
    e = sys.argv[2]
    f = sys.argv[3]
    c(r, e, f)
