import argparse
import shutil
import sys
from multiprocessing import Pool
from multiprocessing.pool import AsyncResult
from pathlib import Path
from git import GitCommandError, Repo
from git.exc import InvalidGitRepositoryError
from loguru import logger
def a(b):
    if not b.exists():
        logger.error(f'Error: {b} does not exist')
        sys.exit(1)
    with b.open() as f:
        c = [d.strip() for d in f if d.strip()]
    if not c:
        logger.error(f'Error: No repositories found in {b}')
        sys.exit(1)
    return c
def e(f):
    g = f.split('/')
    return len(g) == 2 and all(g)
def h(f, i):
    if not e(f):
        return (f, False, f'Invalid format: {f} (expected user/repo)')
    g = f.split('/')
    j = g[0]
    k = g[1]
    l = i / j / k
    if l.exists():
        try:
            Repo(l)
            return (f, True, f'Already exists: {l}')
        except InvalidGitRepositoryError:
            return (f, False, f'Directory exists but is not a git repo: {l}')
    l.parent.mkdir(parents=True, exist_ok=True)
    m = f'https://github.com/{f}.git'
    try:
        Repo.clone_from(m, l, depth=1, single_branch=True)
        return (f, True, f'Successfully cloned to {l}')
    except GitCommandError as e:
        n = str(e).strip()
        if l.exists():
            shutil.rmtree(l, ignore_errors=True)
        return (f, False, f'Clone failed: {n}')
    except Exception as e:
        if l.exists():
            shutil.rmtree(l, ignore_errors=True)
        return (f, False, f'Error: {e!s}')
def o():
    p = argparse.ArgumentParser(description='Clone GitHub repositories in parallel from a text file')
    p.add_argument('file', nargs='?', default='repos.txt', help='Path to file containing repositories (default: repos.txt)')
    p.add_argument('-o', '--output', default='repos', help='Output directory for cloned repositories (default: repos)')
    p.add_argument('--dry-run', action='store_true', help='Show what would be cloned without actually cloning')
    q = p.parse_args()
    r = Path(q.file)
    s = Path(q.output)
    c = a(r)
    print(f'Found {len(c)} repositories to clone')
    if q.dry_run:
        print('\nDry run - would clone:')
        for f in c:
            if e(f):
                g = f.split('/')
                j = g[0]
                k = g[1]
                t = s / j / k
                u = 'EXISTS' if t.exists() else 'NEW'
                print(f'  [{u}] {f} -> {t}')
            else:
                print(f'  [INVALID] {f}')
        return
    v = 0
    w = 0
    x = 0
    print(f'\nCloning with 8 parallel workers to {s.absolute()}')
    print('-' * 40)
    with Pool(processes=8) as y:
        z = [y.apply_async(h, (f, s)) for f in c]
        for f, a1 in zip(c, z):
            try:
                _, b1, c1 = a1.get()
                if b1:
                    if 'Already exists' in c1:
                        x += 1
                        print(f'⏭️  {f}: {c1}')
                    else:
                        v += 1
                        print(f'✅ {f}: {c1}')
                else:
                    w += 1
                    print(f'❌ {f}: {c1}')
            except Exception as e:
                w += 1
                logger.error(f'❌ {f}: Unexpected error: {e!s}')
    print('-' * 40)
    print('\nSummary:')
    print(f'  ✅ Successfully cloned: {v}')
    print(f'  ⏭️  Already existed: {x}')
    print(f'  ❌ Failed: {w}')
    print(f'  📊 Total: {len(c)}')
if __name__ == '__main__':
    raise SystemExit(o())
