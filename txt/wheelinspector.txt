import argparse
import shutil
import zipfile
from pathlib import Path
def a(b):
    try:
        with zipfile.ZipFile(b, 'r') as c:
            d = c.namelist()
            e = any((f.endswith('.py') for f in d))
            g = any((not f.startswith(('dist-info/', '__pycache__/')) and (not f.endswith('/')) and (not f.endswith('.dist-info/')) for f in d))
            return not (e or g)
    except zipfile.BadZipFile:
        print(f'Error: {b} is not a valid zip file')
        return False
    except Exception as e:
        print(f'Error reading {b}: {e}')
        return False
def h(i, j='empty_wheels'):
    k = Path(i)
    l = k / j
    l.mkdir(exist_ok=True)
    m = list(k.glob('*.whl'))
    if not m:
        print(f'No .whl files found in {i}')
        return
    print(f'Found {len(m)} wheel files to check')
    n = []
    o = []
    for p in m:
        print(f'Checking {p.name}...', end=' ')
        if a(p):
            print('EMPTY')
            n.append(p)
        else:
            print('OK')
            o.append(p)
    if not n:
        print('\nNo empty wheels found!')
        return
    print(f'\nFound {len(n)} empty wheel(s)')
    for p in n:
        q = l / p.name
        if q.exists():
            r = 1
            while q.exists():
                q = l / f'{p.stem}_{r}{p.suffix}'
                r += 1
        shutil.move(str(p), str(q))
        print(f'Moved: {p.name} -> {j}/{q.name}')
    print('\nSummary:')
    print(f'  - Empty wheels moved to: {j}/')
    print(f'  - Valid wheels remaining: {len(o)}')
    print(f'  - Total checked: {len(m)}')
def s():
    t = argparse.ArgumentParser(description='Move empty .whl files (only dist-info, no Python code) to a subdirectory')
    t.add_argument('directory', nargs='?', default='.', help='Directory containing .whl files (default: current directory)')
    t.add_argument('-d', '--dest', default='empty_wheels', help="Destination subdirectory name (default: 'empty_wheels')")
    t.add_argument('-v', '--verbose', action='store_true', help='Show detailed output')
    u = t.parse_args()
    v = Path(u.directory)
    if not v.exists():
        print(f"Error: Directory '{u.directory}' does not exist")
        return
